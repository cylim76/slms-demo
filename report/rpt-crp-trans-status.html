version: slms-claim-with-bl-schema-v1
domain: slms
module: container_claim_with_booking

entities:
  booking_crp:
    note: "本页左侧“提箱单（CRP）”。历史页面显示“提箱单号/提单号”，口径统一为 bookingNo。"
    primary_key: bookingNo
    fields:
      bookingNo: { type: string, required: true }   # 页面显示：提箱单号（提单号）
      carrier:   { type: string, required: true }   # 船公司
      vesselVoy: { type: string, required: true }   # 航次（示例中是 vessel 字段）
      etd:       { type: date,   required: true }
      vendor:    { type: string, required: true }   # 运输公司
      pod:       { type: string, required: true }
      country:   { type: string, required: true }
      quota:
        gp20: { type: int, min: 0 }
        gp40: { type: int, min: 0 }
        hq40: { type: int, min: 0 }
      claimed:
        gp20: { type: int, min: 0 }
        gp40: { type: int, min: 0 }
        hq40: { type: int, min: 0 }

  container:
    note: "本页右侧“箱号明细”。"
    primary_key: containerNo
    fields:
      containerNo: { type: string, required: true }
      releaseDate: { type: date, required: true }   # 提箱日期
      vendor:      { type: string, required: true } # 运输公司
      carrier:     { type: string, required: true } # 船公司
      pod:         { type: string, required: true }
      boxType:     { type: enum, required: true, values: ["20GP","40GP","40HQ"] }
      bookingNo:   { type: string, required: false } # 已配箱才有
      claimState:  { type: enum, required: true, values: ["unclaimed","claimed"], default: "unclaimed" }

rules:
  atomic_bind:
    all_or_nothing: true
    validations:
      - "selected containers must all be unclaimed"
      - "container.carrier == booking.carrier"
      - "container.pod == booking.pod"
      - "booking.claimed[type] + need[type] <= booking.quota[type]"
  unbind:
    effect:
      - "container.claimState=unclaimed, container.bookingNo=''"
      - "booking.claimed[type] -= 1 (min 0)"
slms-data-dict.md 字段/口径说明（Data Dictionary） 
# SLMS - 提箱单（CRP）配箱（container-claim-with-bl）

## 术语
- “配箱”= 给提箱单（CRP/Booking）绑定箱号（containerNo）
- 状态：
  - unclaimed = 未配箱
  - claimed   = 已配箱
- 提箱单号/提单号：页面历史显示“提箱单号（提单号）”，口径统一为 bookingNo

## 关键字段口径
### 左侧：提箱单（Booking/CRP）
- quota：计划箱量配额（按箱型）
- claimed：已配箱数量（按箱型）
- 显示：claimed/quota（例：1/3）

### 右侧：箱号明细（Container）
- bookingNo：已配箱时绑定的提箱单号；未配箱为空
- claimState：配箱状态（claimed/unclaimed）

## 计数回退规则
- 解绑时：按箱号的 boxType 对应 booking.claimed[type] -1（最小为 0）
- 配箱时：按选中箱号数量对 booking.claimed[type] +1

## Tab 标题同步
- 选中提箱单后：Tab 标题变更为 “提箱单（CRP）配箱 - {bookingNo}”
- 未选中：Tab 标题为 “提箱单（CRP）配箱”
slms-ui-spec.md 页面交互/约束（UI Spec） 
# SLMS - 提箱单（CRP）配箱 UI Spec

## 1) 布局
- 左侧：提箱单（CRP）查询 + 明细列表 + 抬头详情
- 右侧：箱号明细（配箱对象）查询 + 列表 + 日志
- 中间分割线支持拖动调整左右宽度；记住上次比例（localStorage）

## 2) 列表滚动
- 左侧、右侧列表都必须支持：纵向滚动 + 横向滚动（列多时可左右滑动）

## 3) 选中联动
- 右侧点击某一箱号行：
  - 行高亮
  - 若该箱号已配箱，则自动定位并高亮左侧对应提箱单

## 4) 配箱并绑定（原子操作）
- 前置条件：
  - 必须先选中一个提箱单
  - 必须勾选一个或多个箱号
- 校验（收集全部错误，一次性提示；任何失败则全部不操作）：
  1) 所选箱号均为未配箱（claimed 必须先解绑）
  2) 船公司一致（箱号.carrier == 提箱单.carrier）
  3) 目的港一致（箱号.pod == 提箱单.pod）
  4) 箱型配额足够（claimed + need <= quota）
- 成功：
  - 逐箱号写入 bookingNo，并将状态置为 claimed
  - 同步增加提箱单 claimed 计数
  - 刷新左右列表与抬头详情

## 5) 解绑
- 入口：右侧按钮区“解绑”
- 弹窗确认：列出将解绑的箱号清单（含原提箱单号）
- 成功：
  - 清空箱号 bookingNo，状态改为 unclaimed
  - 回退提箱单 claimed 计数（最小 0）

## 6) 编辑
- 每行“编辑”按钮打开 container-edit.html
- 编辑保存后：本页自动刷新并高亮该箱号（支持 postMessage / storage 事件回传）

## 7) 版本与规范
- 右上角版本号与“三件套弹窗”内容必须同步维护（本页 v7）
