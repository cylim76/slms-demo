# --- Vessel Master (船舶主档) ---
# 演示版存储映射：localStorage key = slms_vessels_v1
# 正式落库建议表名：slms_vessels

tables:
  slms_vessels:
    pk: id
    columns:
      id: {type: string, required: true, note: "主键：VES-xxx"}
      name: {type: string, required: true, note: "船名(EN)，必填"}
      imo: {type: string, required: true, note: "IMO 7位数字，强唯一"}
      mmsi: {type: string, required: false}
      callSign: {type: string, required: false}
      flag: {type: string, required: false}
      type: {type: string, required: false, enum: [Container, Bulk, RoRo]}
      status: {type: string, required: true, enum: [active, inactive], note: "inactive=软删除"}
      remark: {type: string, required: false}
      createdAt: {type: int64, required: true, note: "毫秒时间戳"}
      updatedAt: {type: int64, required: true, note: "毫秒时间戳"}
    uniques:
      - name: uq_slms_vessels_imo
        columns: [imo]
    indexes:
      - name: ix_slms_vessels_status_updatedAt
        columns: [status, updatedAt]

# Vessel Master 字段口径（slms_vessels）

- id：主键（演示 VES-时间戳；正式可用UUID/雪花ID）
- name：船名(EN)，必填
- imo：IMO号（7位数字），必填且全局唯一
- mmsi / callSign / flag / type / remark：可选辅助字段
- status：
  - active：启用（可被航次引用、下拉显示）
  - inactive：停用=软删除（禁止新引用；历史引用保留显示）
- createdAt/updatedAt：毫秒时间戳（演示版）；正式可用datetime

校验口径：
- IMO 必须为 7 位数字
- IMO 强唯一：同一 IMO 只能存在一条记录（不区分状态）

# vessel-manage.html 交互规范（vessel-manage-v3）

入口/导航：
- Index 里归属 Master 一级菜单下的二级菜单：Vessel Master（船舶）

查询：
- 关键词：匹配 name 或 imo（不区分大小写）
- 状态：active / inactive / 全部

新增/编辑：
- 必填：name、imo
- 保存校验：
  - IMO 7位数字
  - IMO 全局唯一（排除当前编辑记录）

停用/启用（软删除）：
- 切换状态需二次确认
- 停用后：不删除记录；仅状态变更 & updatedAt 更新
- 停用后不应出现在“航次管理”等引用下拉中（仅显示 active）

导出：
- 导出当前筛选结果 CSV
- 字段：name, imo, mmsi, callSign, flag, type, status, updatedAt

存储：
- localStorage: slms_vessels_v1（演示）
- 正式：slms_vessels 表（以 schema 为准）

版本维护要求：
- 每次页面生成/修改：必须同步更新
  1) 右上角版本号显示
  2) 版本号弹窗中的“三件套”内容（schema/dict/ui）
